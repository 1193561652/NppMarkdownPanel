cmake_minimum_required(VERSION 3.16)
project(NppMarkdownPanelQt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets)
add_library(NppMarkdownPanelQt MODULE qt/NppMarkdownPanelQt.cpp qt/MarkdownCore.cpp qt/MarkdownCore.h qt/PluginInterface.h)
target_compile_definitions(NppMarkdownPanelQt PRIVATE NPP_PLUGIN_BUILD)
target_link_libraries(NppMarkdownPanelQt PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Widgets)
set_target_properties(NppMarkdownPanelQt PROPERTIES PREFIX "" OUTPUT_NAME "NppMarkdownPanel-qt")
enable_testing()
add_executable(nppmarkdownpanel-qt-tests qt/Tests.cpp qt/MarkdownCore.cpp qt/PluginInterface.h)
target_compile_definitions(nppmarkdownpanel-qt-tests PRIVATE NPP_MARKDOWN_PANEL_PATH="$<TARGET_FILE:NppMarkdownPanelQt>")
target_link_libraries(nppmarkdownpanel-qt-tests PRIVATE Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Widgets)
add_dependencies(nppmarkdownpanel-qt-tests NppMarkdownPanelQt)
add_test(NAME nppmarkdownpanel-qt COMMAND nppmarkdownpanel-qt-tests)
get_target_property(QT_QMAKE_EXECUTABLE Qt${QT_VERSION_MAJOR}::qmake IMPORTED_LOCATION)
get_filename_component(QT_RUNTIME_DIR "${QT_QMAKE_EXECUTABLE}" DIRECTORY)
if(WIN32)
    set_tests_properties(nppmarkdownpanel-qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;PATH=${QT_RUNTIME_DIR}\\;$ENV{PATH}")
else()
    set_tests_properties(nppmarkdownpanel-qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS NppMarkdownPanelQt LIBRARY DESTINATION . RUNTIME DESTINATION .)
